    <h2>Transpositor de Acordes</h2>

   

    <label for="chords">Escribe tus acordes (ej. Sol Mim Lam Re7 o G Em Am D7):</label>

    <textarea id="chords" rows="3">Sol  Mim  Lam  Re7</textarea>

   

    <label for="fromKey">Tonalidad actual (Original):</label>

    <select id="fromKey"></select>

   

    <label for="toKey">Nueva tonalidad (Destino):</label>

    <select id="toKey"></select>

    <label for="outputFormat">Formato de salida:</label>

    <select id="outputFormat">

      <option value="es">Español (Do, Re, Mi...)</option>

      <option value="en">Americano (C, D, E...)</option>

    </select>

   

    <button onclick="transpose()">Transponer Acordes</button>

   

    <label>Resultado:</label>

    <div class="result" id="output">Do  Lam  Rem  Sol7</div>

  </div>

  <script>

    const notesES = ["Do", "Do#", "Re", "Re#", "Mi", "Fa", "Fa#", "Sol", "Sol#", "La", "La#", "Si"];

    const notesEN = ["C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"];

   

    const noteMap = {

      'C':0, 'DO':0,

      'C#':1, 'DB':1, 'DO#':1, 'REB':1,

      'D':2, 'RE':2,

      'D#':3, 'EB':3, 'RE#':3, 'MIB':3,

      'E':4, 'MI':4,

      'F':5, 'FA':5,

      'F#':6, 'GB':6, 'FA#':6, 'SOLB':6,

      'G':7, 'SOL':7,

      'G#':8, 'AB':8, 'SOL#':8, 'LAB':8,

      'A':9, 'LA':9,

      'A#':10, 'BB':10, 'LA#':10, 'SIB':10,

      'B':11, 'SI':11

    };

    const fromSelect = document.getElementById('fromKey');

    const toSelect = document.getElementById('toKey');

   

    notesES.forEach((note, i) => {

      fromSelect.options.add(new Option(`${note} / ${notesEN[i]}`, i));

      toSelect.options.add(new Option(`${note} / ${notesEN[i]}`, i));

    });

   

    fromSelect.selectedIndex = 7; // Sol / G

    toSelect.selectedIndex = 0;   // Do / C

    function transpose() {

      const from = parseInt(fromSelect.value);

      const to = parseInt(toSelect.value);

      const shift = (to - from + 12) % 12;

      const format = document.getElementById('outputFormat').value;

      const targetArray = format === 'es' ? notesES : notesEN;

     

      const input = document.getElementById('chords').value;

     

      // Regex busca notas de 2 o 3 letras primero para evitar coincidencias parciales

      const regex = /(Sol#|Solb|Sol|Do#|Reb|Do|Re#|Mib|Re|Mi|Fa#|Solb|Fa|La#|Sib|Lab|La|Si|C#|Db|C|D#|Eb|D|E|F#|Gb|F|G#|Ab|G|A#|Bb|A|B)/gi;

     

      const transposed = input.replace(regex, (match) => {

        const key = match.toUpperCase();

        if (noteMap[key] !== undefined) {

          const originalIdx = noteMap[key];

          const newIdx = (originalIdx + shift) % 12;

          return targetArray[newIdx];

        }

        return match;

      });

     

      document.getElementById('output').innerText = transposed;

    }

  </script>
